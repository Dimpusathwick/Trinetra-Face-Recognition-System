-- MySQL schema for Face ID (identities + embeddings)
-- Run this SQL in your MySQL server (database: faceid)

CREATE DATABASE IF NOT EXISTS faceid DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE faceid;

-- identities table: stores logical person records
CREATE TABLE IF NOT EXISTS identities (
    id VARCHAR(64) NOT NULL PRIMARY KEY,
    name VARCHAR(191) NOT NULL UNIQUE,
    embedding_id VARCHAR(64) NULL,
    metadata JSON NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- embeddings table: stores multiple embeddings per identity
CREATE TABLE IF NOT EXISTS embeddings (
    id VARCHAR(64) NOT NULL PRIMARY KEY,
    vector JSON NOT NULL,
    quality FLOAT NULL,
    source_id VARCHAR(64) NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_source_id (source_id)
);
